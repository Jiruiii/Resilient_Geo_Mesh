```kotlin
package com.resilientgeo.mesh.trust

import java.math.BigDecimal
import java.math.RoundingMode

/** A location hint from the offline address search; not an authoritative address. */
data class CrowdLocationHint(
    val method: String,
    val query: String,
    val label: String,
    val kind: String,
    val precision: String,
)

/**
 * Kotlin port of `pipeline/lib/crowd-common.mjs`, shared by crowd reports and
 * rescue requests. Error strings match the JS module verbatim.
 */
object CrowdCommon {
    val LOCATION_SOURCES = setOf("CURRENT_LOCATION", "MAP_PICK")
    val HINT_KINDS = setOf("COUNTY", "DISTRICT", "VILLAGE", "ROAD", "FACILITY")
    val HINT_PRECISIONS = setOf("AREA", "ROAD", "POINT")
    const val HINT_TEXT_MAX_CODE_POINTS = 80

    /** [minLon, minLat, maxLon, maxLat]: Taiwan incl. Penghu, Kinmen, Matsu and Pratas. */
    val TAIWAN_BOUNDS = doubleArrayOf(116.5, 20.5, 122.5, 26.8)

    /** Matches JS `Number(value.toFixed(6))` for the coordinates the factories accept. */
    fun roundCoordinate(value: Double): Double = BigDecimal(value).setScale(6, RoundingMode.HALF_UP).toDouble()

    fun hintErrors(hint: CrowdLocationHint): List<String> {
        val errors = mutableListOf<String>()
        if (hint.method != "ADDRESS") errors += "location_hint.method must be ADDRESS"
        if (hint.kind !in HINT_KINDS) errors += "location_hint.kind is invalid"
        if (hint.precision !in HINT_PRECISIONS) errors += "location_hint.precision is invalid"
        for ((field, value) in listOf("query" to hint.query, "label" to hint.label)) {
            if (value.codePointCount(0, value.length) > HINT_TEXT_MAX_CODE_POINTS) {
                errors += "location_hint.$field must be a string of at most $HINT_TEXT_MAX_CODE_POINTS characters"
            }
        }
        return errors
    }

    fun coordinateErrors(lon: Double, lat: Double): List<String> {
        if (!lon.isFinite() || !lat.isFinite()) return listOf("location must be finite numbers")
        if (lon < TAIWAN_BOUNDS[0] || lon > TAIWAN_BOUNDS[2] || lat < TAIWAN_BOUNDS[1] || lat > TAIWAN_BOUNDS[3]) {
            return listOf("location is outside the supported area")
        }
        return emptyList()
    }
}
```

```kotlin
package com.resilientgeo.mesh.trust

import org.json.JSONObject
import java.time.Duration

data class RescueRequestInput(
    val requestType: String,
    val subject: String,
    val peopleCount: Int,
    val triage: Map<String, String>,
    val description: String,
    val lon: Double,
    val lat: Double,
    val locationSource: String,
    val locationHint: CrowdLocationHint? = null,
)

/**
 * Kotlin port of `pipeline/lib/rescue-contract.mjs`. Error codes and their
 * order are part of the contract; `RescueContractTest` replays the shared
 * fixture against this object.
 */
object RescueContract {
    const val NAMESPACE = "crowd.rescue"
    const val EVENT_TYPE = "RESCUE_REQUEST"
    const val SOURCE = "CROWD"
    const val SOURCE_VERSION = "rescue-request-v0"
    const val DATASET_ID = "crowd-rescue"
    const val MANIFEST_ID = "crowd-rescue:none"
    val TTL: Duration = Duration.ofHours(6)
    val MAX_LIFETIME: Duration = Duration.ofHours(72)
    const val MAX_EVENT_VERSION = 200
    const val PEOPLE_MAX = 50
    const val DESCRIPTION_MAX_CODE_POINTS = 160

    val REQUEST_TYPES = listOf("FIRST_AID_NEEDED", "IMMOBILE", "TRAPPED", "TRAPPED_DEBRIS", "OTHER_RESCUE")
    val STATUSES = listOf("ACTIVE", "RESOLVED", "CANCELLED")
    val SUBJECTS = listOf("SELF", "OTHERS")
    val TRIAGE_FIELDS = listOf("immobile", "breathing_difficulty", "severe_bleeding", "conscious")
    val TRIAGE_ANSWERS = setOf("YES", "NO", "UNKNOWN")

    private val ATTRIBUTE_KEYS = setOf(
        "request_type", "status", "subject", "people_count", "triage", "description",
        "location_source", "location_hint", "first_issued_at", "area_id", "theme",
    )
    private val EVENT_ID_RE =
        Regex("^rescue:([0-9a-f]{32}):[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$")

    fun deriveUrgency(requestType: String?, triage: Map<String, Any?>): String = when {
        requestType == "TRAPPED_DEBRIS" -> "CRITICAL"
        triage["breathing_difficulty"] == "YES" || triage["severe_bleeding"] == "YES" || triage["conscious"] == "NO" -> "CRITICAL"
        requestType == "TRAPPED" || requestType == "IMMOBILE" || triage["immobile"] == "YES" -> "HIGH"
        else -> "MEDIUM"
    }

    fun eventIdPrefix(signingKeyId: String): String = "rescue:${signingKeyId.removePrefix(DeviceKeys.PREFIX)}:"

    fun isTerminal(status: String?): Boolean = status == "RESOLVED" || status == "CANCELLED"

    private fun triageValid(triage: Map<String, Any?>, subject: String?): Boolean {
        if (triage.keys != TRIAGE_FIELDS.toSet()) return false
        if (!TRIAGE_FIELDS.all { triage[it] is String && triage[it] in TRIAGE_ANSWERS }) return false
        return subject != "SELF" || triage["conscious"] == "YES"
    }

    private fun codePoints(value: String): Int = value.codePointCount(0, value.length)

    fun fieldErrors(input: RescueRequestInput): List<String> = fieldErrors(
        requestType = input.requestType,
        subject = input.subject,
        peopleCount = input.peopleCount,
        triage = input.triage,
        description = input.description,
        lon = input.lon,
        lat = input.lat,
        locationSource = input.locationSource,
        hintValid = input.locationHint?.let { CrowdCommon.hintErrors(it).isEmpty() } ?: true,
    )

    private fun fieldErrors(
        requestType: String?,
        subject: String?,
        peopleCount: Int?,
        triage: Map<String, Any?>?,
        description: String?,
        lon: Double,
        lat: Double,
        locationSource: String?,
        hintValid: Boolean,
    ): List<String> {
        val errors = mutableListOf<String>()
        if (requestType !in REQUEST_TYPES) errors += "rescue_field_invalid:request_type"
        if (subject !in SUBJECTS) errors += "rescue_field_invalid:subject"
        if (peopleCount == null || peopleCount < 1 || peopleCount > PEOPLE_MAX) errors += "rescue_field_invalid:people_count"
        if (triage == null || !triageValid(triage, subject)) errors += "rescue_field_invalid:triage"
        if (description == null || codePoints(description.trim()) > DESCRIPTION_MAX_CODE_POINTS) {
            errors += "rescue_field_invalid:description"
        }
        if (locationSource !in CrowdCommon.LOCATION_SOURCES) errors += "rescue_field_invalid:location_source"
        if (!hintValid) errors += "rescue_field_invalid:location_hint"
        if (CrowdCommon.coordinateErrors(lon, lat).isNotEmpty()) errors += "rescue_geometry_invalid"
        return errors
    }

    /** Stateless rules; assumes EventShapeValidator already passed. Returns at most one code. */
    fun eventErrors(event: JSONObject): List<String> {
        val attributes = event.optJSONObject("attributes") ?: JSONObject()
        if (event.optString("namespace") != NAMESPACE || event.optString("event_type") != EVENT_TYPE ||
            event.optString("source") != SOURCE || event.optString("source_version") != SOURCE_VERSION ||
            attributes.opt("area_id") != "crowd" || attributes.opt("theme") != "rescue"
        ) {
            return listOf("rescue_identity_invalid")
        }
        val match = EVENT_ID_RE.matchEntire(event.optString("event_id"))
        if (match == null || DeviceKeys.PREFIX + match.groupValues[1] != event.optString("signing_key_id")) {
            return listOf("rescue_event_id_unbound")
        }
        val geometry = event.optJSONObject("geometry")
        val coordinates = geometry?.optJSONArray("coordinates")
        if (geometry?.optString("type") != "Point" || coordinates == null || coordinates.length() != 2) {
            return listOf("rescue_geometry_invalid")
        }
        if (attributes.keys().asSequence().any { it !in ATTRIBUTE_KEYS }) return listOf("rescue_attributes_unknown")

        val triageJson = attributes.opt("triage") as? JSONObject
        val triage = triageJson?.let { json -> json.keys().asSequence().associateWith { json.get(it) } }
        val hint = attributes.opt("location_hint")
        val hintValid = when (hint) {
            null -> true
            is JSONObject -> {
                val fields = listOf("method", "query", "label", "kind", "precision")
                fields.all { hint.opt(it) is String } && CrowdCommon.hintErrors(
                    CrowdLocationHint(
                        hint.getString("method"), hint.getString("query"), hint.getString("label"),
                        hint.getString("kind"), hint.getString("precision"),
                    ),
                ).isEmpty()
            }
            else -> false
        }
        val people = when (val value = attributes.opt("people_count")) {
            is Int -> value
            is Long -> value.takeIf { it in Int.MIN_VALUE..Int.MAX_VALUE }?.toInt()
            else -> null
        }
        val fieldErrors = fieldErrors(
            requestType = attributes.opt("request_type") as? String,
            subject = attributes.opt("subject") as? String,
            peopleCount = people,
            triage = triage,
            description = attributes.opt("description") as? String,
            lon = (coordinates.opt(0) as? Number)?.toDouble() ?: Double.NaN,
            lat = (coordinates.opt(1) as? Number)?.toDouble() ?: Double.NaN,
            locationSource = attributes.opt("location_source") as? String,
            hintValid = hintValid,
        )
        if (fieldErrors.isNotEmpty()) return listOf(fieldErrors.first())
        if (attributes.opt("status") !in STATUSES) return listOf("rescue_field_invalid:status")
        if (event.optString("severity") != deriveUrgency(attributes.optString("request_type"), triage!!)) {
            return listOf("rescue_severity_mismatch")
        }
        val version = event.optInt("event_version")
        if (version > MAX_EVENT_VERSION) return listOf("rescue_version_too_high")
        val issued = EventShapeValidator.parseTimeOrNull(event.optString("issued_at"))
        val expires = EventShapeValidator.parseTimeOrNull(event.optString("expires_at"))
        val first = EventShapeValidator.parseTimeOrNull(attributes.opt("first_issued_at") as? String)
        if (issued == null || expires == null || first == null || first.isAfter(issued) || !expires.isAfter(issued) ||
            (version == 1 && first != issued)
        ) {
            return listOf("rescue_time_invalid")
        }
        if (Duration.between(issued, expires) > TTL) return listOf("rescue_ttl_too_long")
        if (expires.isAfter(first.plus(MAX_LIFETIME))) return listOf("rescue_lifetime_exceeded")
        return emptyList()
    }

    fun transitionErrors(stored: JSONObject, incoming: JSONObject): List<String> {
        if (stored.optString("signing_key_id") != incoming.optString("signing_key_id")) return listOf("signer_mismatch")
        val storedAttributes = stored.optJSONObject("attributes")
        val incomingAttributes = incoming.optJSONObject("attributes")
        if (isTerminal(storedAttributes?.optString("status"))) return listOf("rescue_terminal")
        if (isTerminal(incomingAttributes?.optString("status")) &&
            incoming.optString("expires_at") != stored.optString("expires_at")
        ) {
            return listOf("rescue_terminal_expiry_changed")
        }
```

```kotlin
package com.resilientgeo.mesh.report

import com.resilientgeo.mesh.trust.CrowdCommon
import com.resilientgeo.mesh.trust.CrowdLocationHint
import com.resilientgeo.mesh.trust.DeviceSigningKey
import com.resilientgeo.mesh.trust.RescueContract

import org.json.JSONArray
import org.json.JSONObject
import java.time.Instant
import java.time.temporal.ChronoUnit
import java.util.UUID

enum class RescueAction { UPDATE, EXTEND, RESOLVE, CANCEL }

/**
 * Kotlin port of `buildRescueRequest()` / `buildRescueRevision()` in
 * `pipeline/lib/rescue-request.mjs`; `RescueRequestFactoryTest` rebuilds the
 * shared fixture timeline byte for byte.
 */
object RescueRequestFactory {

    fun create(input: RescueRequestInput, key: DeviceSigningKey, issuedAt: Instant, requestUuid: UUID = UUID.randomUUID()): JSONObject {
        val issued = issuedAt.truncatedTo(ChronoUnit.SECONDS)
        return assembleAndSign(
            input = input,
            eventId = RescueContract.eventIdPrefix(key.keyId()) + requestUuid.toString().lowercase(),
            version = 1,
            issued = issued,
            expires = issued.plus(RescueContract.TTL),
            status = "ACTIVE",
            firstIssuedAt = issued.toString(),
            key = key,
        )
    }

    fun revise(previous: JSONObject, action: RescueAction, changes: RescueRequestInput?, key: DeviceSigningKey, issuedAt: Instant): JSONObject {
        require(key.keyId() == previous.getString("signing_key_id")) { "rescue_revision_wrong_device" }
        val previousAttributes = previous.getJSONObject("attributes")
        require(!RescueContract.isTerminal(previousAttributes.getString("status"))) { "rescue_revision_terminal" }
        val issued = issuedAt.truncatedTo(ChronoUnit.SECONDS)
        val previousExpires = Instant.parse(previous.getString("expires_at"))
        require(issued.isBefore(previousExpires)) { "rescue_revision_expired" }
        val firstIssuedAt = previousAttributes.getString("first_issued_at")
        val expires = if (action == RescueAction.EXTEND) {
            val cap = Instant.parse(firstIssuedAt).plus(RescueContract.MAX_LIFETIME)
            val extended = minOf(issued.plus(RescueContract.TTL), cap)
            require(extended.isAfter(previousExpires)) { "rescue_extend_not_later" }
            extended
        } else {
            previousExpires
        }
        val input = if (action == RescueAction.UPDATE) requireNotNull(changes) { "rescue_action_invalid" } else inputFromEvent(previous)
        val status = when (action) {
            RescueAction.RESOLVE -> "RESOLVED"
            RescueAction.CANCEL -> "CANCELLED"
            else -> "ACTIVE"
        }
        return assembleAndSign(
            input = input,
            eventId = previous.getString("event_id"),
            version = previous.getInt("event_version") + 1,
            issued = issued,
            expires = expires,
            status = status,
            firstIssuedAt = firstIssuedAt,
            key = key,
        )
    }

    fun inputFromEvent(event: JSONObject): RescueRequestInput {
        val attributes = event.getJSONObject("attributes")
        val triage = attributes.getJSONObject("triage")
        val coordinates = event.getJSONObject("geometry").getJSONArray("coordinates")
        val hint = attributes.optJSONObject("location_hint")
        return RescueRequestInput(
            requestType = attributes.getString("request_type"),
            subject = attributes.getString("subject"),
            peopleCount = attributes.getInt("people_count"),
            triage = triage.keys().asSequence().associateWith { triage.getString(it) },
            description = attributes.getString("description"),
            lon = coordinates.getDouble(0),
            lat = coordinates.getDouble(1),
            locationSource = attributes.getString("location_source"),
            locationHint = hint?.let {
                CrowdLocationHint(it.getString("method"), it.getString("query"), it.getString("label"), it.getString("kind"), it.getString("precision"))
            },
        )
    }

    private fun normalizedTriage(subject: String, triage: Map<String, String>): Map<String, String> = linkedMapOf(
        "immobile" to triage["immobile"].orEmpty(),
        "breathing_difficulty" to triage["breathing_difficulty"].orEmpty(),
        "severe_bleeding" to triage["severe_bleeding"].orEmpty(),
        "conscious" to if (subject == "SELF") "YES" else triage["conscious"].orEmpty(),
    )

    private fun assembleAndSign(
        input: RescueRequestInput,
        eventId: String,
        version: Int,
        issued: Instant,
        expires: Instant,
        status: String,
        firstIssuedAt: String,
        key: DeviceSigningKey,
    ): JSONObject {
        val normalized = input.copy(triage = normalizedTriage(input.subject, input.triage))
        val errors = RescueContract.fieldErrors(normalized)
        require(errors.isEmpty()) { errors.joinToString("; ") }
        val triage = JSONObject()
        normalized.triage.forEach { (field, answer) -> triage.put(field, answer) }
        val attributes = JSONObject()
            .put("request_type", normalized.requestType)
            .put("status", status)
            .put("subject", normalized.subject)
            .put("people_count", normalized.peopleCount)
            .put("triage", triage)
            .put("description", normalized.description.trim())
            .put("location_source", normalized.locationSource)
            .put("first_issued_at", firstIssuedAt)
            .put("area_id", "crowd")
            .put("theme", "rescue")
        normalized.locationHint?.let { hint ->
            attributes.put(
                "location_hint",
                JSONObject().put("method", hint.method).put("query", hint.query).put("label", hint.label)
                    .put("kind", hint.kind).put("precision", hint.precision),
            )
        }
        val event = JSONObject()
            .put("schema_version", "event-v0")
            .put("namespace", RescueContract.NAMESPACE)
            .put("event_id", eventId)
            .put("event_type", RescueContract.EVENT_TYPE)
            .put(
                "geometry",
                JSONObject().put("type", "Point").put(
                    "coordinates",
                    JSONArray().put(CrowdCommon.roundCoordinate(normalized.lon)).put(CrowdCommon.roundCoordinate(normalized.lat)),
                ),
            )
            .put("severity", RescueContract.deriveUrgency(normalized.requestType, normalized.triage))
            .put("source", RescueContract.SOURCE)
            .put("source_version", RescueContract.SOURCE_VERSION)
            .put("event_version", version)
            .put("issued_at", issued.toString())
            .put("expires_at", expires.toString())
            .put("attributes", attributes)
            .put("signing_key_id", key.keyId())
            .put("signer_public_key", key.publicKeySpkiBase64())
            .put(
                "provenance",
                JSONObject()
                    .put("original_source", "crowd-app")
                    .put("received_at", issued.toString())
                    .put("transport_source", JSONObject().put("kind", "local_report")),
            )
        return CrowdReportFactory.signEvent(event, key)
    }
}
```

```kotlin
sealed class RescueResult {
    data class Saved(val event: EventEntity) : RescueResult()
    data class Invalid(val errors: List<String>) : RescueResult()
    data class Conflict(val reason: String) : RescueResult()
    data class RateLimited(val retryAfterSeconds: Long) : RescueResult()
    data class SigningUnavailable(val message: String) : RescueResult()
    data class StorageUnavailable(val message: String) : RescueResult()
}

internal data class RescueCandidate(
    val chunkId: String,
    val status: String,
    val severity: String,
    val receivedAtEpochMillis: Long,
)

private val RESCUE_SEVERITY_RANK = mapOf("MEDIUM" to 0, "HIGH" to 1, "CRITICAL" to 2)

/** Which relayed rescue requests to drop so at most [keep] remain: terminal first, then least urgent, then oldest. */
internal fun rescueEvictionOrder(candidates: List<RescueCandidate>, keep: Int): List<RescueCandidate> =
    candidates
        .sortedWith(
            compareBy<RescueCandidate>(
                { if (RescueContract.isTerminal(it.status)) 0 else 1 },
                { RESCUE_SEVERITY_RANK[it.severity] ?: 0 },
                { it.receivedAtEpochMillis },
            ),
        )
```
